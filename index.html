TRAVA AUTOMÁTICA EM CI/CD (GITHUB ACTIONS)

              
Bloqueie Pull Requests com nota de segurança inferior a "A" antes do deploy

            

          

          
        


        


name: ObsidianSec DevSecOps Quality Gate

on: [push, pull_request]

jobs:
  security-audit:
    name: Tactical Edge & Secret Gate
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Run Secret & SAST Scan
        run: npx obsidiansec scan-dir ./

      - name: Run Live Edge Quality Gate
        run: npx obsidiansec audit https://staging.myapp.com --min-grade=A
